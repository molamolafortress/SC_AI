cmake_minimum_required(VERSION 3.16)
project(SidecarProbeModule CXX)
set(CMAKE_CXX_STANDARD 17)
# BWAPI_ROOT: the OpenBW bwapi checkout (headers under bwapi/include, libs under build/lib)
set(BWAPI_ROOT "${CMAKE_SOURCE_DIR}/../../../third_party/openbw-bwapi" CACHE PATH "OpenBW bwapi checkout")
set(BWAPI_LIB_DIR "${BWAPI_ROOT}/build/lib" CACHE PATH "built libs")
include_directories(${BWAPI_ROOT}/bwapi/include ${CMAKE_SOURCE_DIR}/..)
set(CMAKE_SHARED_LIBRARY_PREFIX "")
add_library(SidecarProbeModule SHARED ProbeModule.cpp)
find_library(BWAPILIB_SO NAMES BWAPILIB PATHS ${BWAPI_LIB_DIR} NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
target_link_libraries(SidecarProbeModule ${BWAPILIB_SO} Threads::Threads)
